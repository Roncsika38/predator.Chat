CREATE OR REPLACE FUNCTION public.grant_predator_role( p_user_id uuid, p_role text ) RETURNS boolean LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $$ DECLARE owner_id uuid := 'a25e2ffd-5f5c-4972-894d-7bfae6f879b4'; BEGIN -- Csak a tulajdonos kezelheti a szerepköröket. IF (SELECT auth.uid()) IS DISTINCT FROM owner_id THEN RAISE EXCEPTION 'Nincs jogosultságod ehhez a művelethez.'; END IF; -- A tulajdonos saját adminjogát nem lehet ezzel elvenni. IF p_user_id = owner_id THEN RAISE EXCEPTION 'A tulajdonos szerepköre nem módosítható.'; END IF; IF p_role NOT IN ('admin', 'member') THEN RAISE EXCEPTION 'Érvénytelen szerepkör.'; END IF; IF NOT EXISTS ( SELECT 1 FROM auth.users WHERE id = p_user_id ) THEN RAISE EXCEPTION 'Nem található ilyen felhasználó.'; END IF; INSERT INTO public.user_roles (user_id, role) VALUES (p_user_id, p_role) ON CONFLICT (user_id) DO UPDATE SET role = EXCLUDED.role; RETURN true; END; $$; REVOKE ALL ON FUNCTION public.grant_predator_role(uuid, text) FROM PUBLIC, anon; GRANT EXECUTE ON FUNCTION public.grant_predator_role(uuid, text) TO authenticated;
